# 数据处理层模块文档

**文件**: `Software/data_process.py`

## 1. 模块职责

批量处理FFT数据，生成瀑布图，提供窗口切片接口。

## 2. 核心类：DataProcessor

### 2.1 初始化

```python
def __init__(self, state):
    self.state = state
    self.fft_data_queue = None  # 由外部设置
    
    # 扫描参数
    self.channel_count = 20
    self.fft_length = 512
    self.total_fft_length = 10240  # 20×512
    self.total_bandwidth_mhz = 2000
    
    # 瀑落图buffer
    self.waterfall_height = state.waterfall_height  # 默认512
    self.waterfall_width = 10240
    zero_line = np.zeros(10240, dtype=np.float32)
    self.waterfall_buffer = deque([zero_line] * 512, maxlen=512)
    
    # RGB图像
    self.waterfall_image = np.zeros((10240, 512, 3), dtype=np.uint8)
    
    # 线程锁
    self.data_lock = threading.Lock()
    self.image_lock = threading.Lock()
```

## 3. 核心方法

### 3.1 线程管理

#### start_processing()
```python
def start_processing(self):
    """启动双线程处理"""
    # 处理线程：更新buffer
    self.process_thread = threading.Thread(target=self._process_loop, daemon=True)
    self.process_thread.start()
    
    # 转换线程：生成RGB
    self.image_thread = threading.Thread(target=self._image_conversion_loop, daemon=True)
    self.image_thread.start()
```

#### stop_processing()
```python
def stop_processing(self):
    """停止处理线程"""
    self.state.data_processing_thread = False
    self.process_thread.join(timeout=2)
    self.image_thread.join(timeout=2)
```

### 3.2 数据处理（线程1）

#### _process_loop()
```python
def _process_loop(self):
    """处理线程主循环"""
    while self.state.data_processing_thread:
        # 1. 批量获取队列数据
        batch_frames = []
        first = self.fft_data_queue.get(timeout=1)
        batch_frames.append(first)
        
        while True:
            try:
                batch_frames.append(self.fft_data_queue.get_nowait())
            except queue.Empty:
                break
        
        # 2. 批量处理
        for fft_frame in batch_frames:
            fft_data = fft_frame["data"]  # ndarray(10240,)
            
            # 确保长度正确
            if len(fft_data) != 10240:
                fft_data = fft_data[:10240]  # 截断
            
            # (可选) 噪声滤波
            if self.enable_noise_filter:
                fft_data = self._apply_noise_filter(fft_data)
            
            # 更新buffer (保存原始dB值)
            with self.data_lock:
                self.waterfall_buffer.append(fft_data)
                self.latest_spectrum = fft_data.copy()
        
        self.image_needs_update = True
```

#### 噪声滤波（可选）
```python
def _apply_noise_filter(self, fft_data):
    """噪声滤波"""
    if self.noise_filter_mode == "subtraction":
        # 动态噪声底估计
        if self.noise_floor is None:
            self.noise_floor = fft_data.copy()
        else:
            # 非对称更新
            lower_mask = fft_data < self.noise_floor
            self.noise_floor[lower_mask] = (
                (1 - self.noise_alpha) * self.noise_floor[lower_mask] +
                self.noise_alpha * fft_data[lower_mask]
            )
            self.noise_floor[~lower_mask] = (
                (1 - self.noise_alpha/10) * self.noise_floor[~lower_mask] +
                (self.noise_alpha/10) * fft_data[~lower_mask]
            )
        
        # 减法
        return np.maximum(fft_data - self.noise_floor, 0)
    
    elif self.noise_filter_mode == "threshold":
        # 硬阈值
        threshold = np.percentile(fft_data, 10)
        return np.where(fft_data > threshold, fft_data, 0)
```

### 3.3 图像转换（线程2）

#### _image_conversion_loop()
```python
def _image_conversion_loop(self):
    """图像转换线程"""
    while self.state.data_processing_thread:
        if not self.image_needs_update:
            time.sleep(0.01)
            continue
        
        # 1. 获取buffer (线程安全)
        with self.data_lock:
            waterfall_list = list(self.waterfall_buffer)
            self.image_needs_update = False
        
        # 2. 全局归一化
        waterfall_array = np.array(waterfall_list)  # (512, 10240)
        min_db = np.min(waterfall_array)
        max_db = np.max(waterfall_array)
        normalized = (waterfall_array - min_db) / (max_db - min_db + 1e-12)
        
        # 3. 转置+翻转
        normalized = np.flipud(normalized).T  # (10240, 512)
        
        # 4. 色图映射
        gray = (normalized * 255).astype(np.uint8)
        bgr_image = cv2.applyColorMap(gray, cv2.COLORMAP_JET)
        
        # 5. 保存
        with self.image_lock:
            self.waterfall_image = bgr_image
            self.max_value = float(max_db)
            self.min_value = float(min_db)
```

**关键点**：
- **全局归一化**：对整个buffer统一缩放，避免逐帧归一化的亮度闪烁
- **转置翻转**：`flipud().T` 使时间轴从下到上，频率轴从左到右
- **BGR格式**：OpenCV输出BGR，Qt显示时需转换

### 3.4 窗口切片（关键接口）

#### get_window_image(start_point, end_point)
```python
def get_window_image(self, start_point, end_point):
    """获取指定FFT点范围的图像切片
    
    Args:
        start_point: 起始FFT点 (0-10239)
        end_point: 结束FFT点 (1-10240)
    
    Returns:
        ndarray: BGR图像 (end-start, height, 3)
    """
    # 边界检查
    start_point = max(0, min(start_point, 10239))
    end_point = max(start_point + 1, min(end_point, 10240))
    
    with self.image_lock:
        sliced = self.waterfall_image[start_point:end_point, :, :].copy()
    
    return sliced
```

**使用场景**：
- 扫描模式：`get_window_image(window_start, window_end)`
- 固定显示：`get_window_image(freq_to_point(f1), freq_to_point(f2))`

#### get_point_to_frequency(point_index)
```python
def get_point_to_frequency(self, point_index):
    """FFT点 → 频率(MHz)"""
    return point_index * 2000 / 10240
```

**反向转换**：
```python
def frequency_to_point(freq_mhz):
    return int(freq_mhz * 10240 / 2000)
```

## 4. 数据流

```
fft_data_queue
  ↓ batch获取
_process_loop()
  ↓ 批量处理
waterfall_buffer (deque)
  ↓ 标记更新
_image_conversion_loop()
  ↓ 全局归一化
cv2.applyColorMap()
  ↓ BGR图像
waterfall_image
  ↓ 切片
get_window_image(start, end)
  ↓
ScanningController/UI
```

## 5. 线程安全

| 共享资源 | 保护锁 | 访问线程 |
|---------|-------|---------|
| `waterfall_buffer` | `data_lock` | 处理线程(写) + 转换线程(读) |
| `waterfall_image` | `image_lock` | 转换线程(写) + 外部(读) |
| `latest_spectrum` | `data_lock` | 处理线程(写) + 外部(读) |

## 6. 配置接口

### set_fft_length(length)
```python
def set_fft_length(self, length):
    """更新FFT长度（运行时）"""
    with self.data_lock:
        self.fft_length = length
        self.total_fft_length = length * 20
        self.waterfall_width = self.total_fft_length
        
        # 重置buffer
        zero = np.zeros(self.waterfall_width, dtype=np.float32)
        self.waterfall_buffer = deque([zero] * self.waterfall_height, 
                                      maxlen=self.waterfall_height)
```

### set_noise_filter_parameters(enable, mode, alpha)
```python
def set_noise_filter_parameters(self, enable, mode=None, alpha=None):
    """配置噪声滤波
    
    Args:
        enable: 是否启用
        mode: "subtraction" | "threshold"
        alpha: 学习率 (0-1)
    """
    with self.data_lock:
        self.enable_noise_filter = enable
        if mode:
            self.noise_filter_mode = mode
        if alpha:
            self.noise_alpha = max(0, min(1, alpha))
        
        if not enable:
            self.noise_floor = None  # 重置
```

## 7. 性能优化

### 7.1 批量处理
```python
# ❌ 低效：逐帧处理
for _ in range(N):
    frame = queue.get()
    process(frame)

# ✅ 高效：批量处理
batch = [queue.get()]
while True:
    try:
        batch.append(queue.get_nowait())
    except queue.Empty:
        break
process_batch(batch)
```

### 7.2 全局归一化
```python
# ❌ 逐帧归一化 → 亮度闪烁
for line in buffer:
    normalized = (line - line.min()) / (line.max() - line.min())

# ✅ 全局归一化 → 稳定显示
array = np.array(buffer)
global_min, global_max = array.min(), array.max()
normalized = (array - global_min) / (global_max - global_min)
```

### 7.3 OpenCV色图
```python
# OpenCV COLORMAP_JET 比 matplotlib 快10倍
gray = (normalized * 255).astype(np.uint8)
bgr = cv2.applyColorMap(gray, cv2.COLORMAP_JET)
```
