### 2.2 模型加载

#### _load_model()
```python
def _load_model(self):
    """优先OpenVINO GPU，回退PyTorch CPU"""
    try:
        # 尝试OpenVINO + GPU
        core = Core()
        if "GPU" in core.available_devices:
            self.model = YOLO(str(self.openvino_model_path))
            self.use_openvino = True
            logger.info("✓ OpenVINO GPU模式")
            return
    except:
        pass
    
    # 回退PyTorch
    self.model = YOLO(str(self.model_path))
    self.use_openvino = False
    logger.info("✓ PyTorch CPU模式")
```

**性能对比**：
- OpenVINO GPU: ~50-80 FPS
- PyTorch CPU: ~10-20 FPS

### 2.3 检测循环

#### _detection_loop()
```python
def _detection_loop(self):
    """检测主循环"""
    while self.state.detection_thread:
        start_time = time.time()
        
        # 1. 从扫描控制器获取当前窗口
        image = self.scanning_controller.get_current_window_image()
        
        # 2. YOLO检测
        detections = self._detect(image)
        
        # 3. 更新状态机
        self.scanning_controller.update_state_machine(
            detections, 
            image.shape[1]  # 图像宽度
        )
        
        # 4. 绘制检测框
        annotated = self._draw_detections(image, detections)
        
        # 5. 保存结果
        with self.detection_lock:
            self.detection_image = annotated
            self.detection_results = detections
            self.total_objects += len(detections)
        
        # 6. 计算FPS
        self.fps = 1.0 / (time.time() - start_time)
```

#### _detect(image)
```python
def _detect(self, image):
    """执行YOLO检测"""
    kwargs = {
        "conf": self.conf_threshold,
        "iou": self.iou_threshold,
        "verbose": False
    }
    
    if self.use_openvino:
        kwargs["device"] = "intel:gpu"
        kwargs["imgsz"] = 512
    else:
        kwargs["imgsz"] = self.image_size
    
    results = self.model(image, **kwargs)
    
    # 解析结果
    detections = []
    if len(results) > 0 and results[0].boxes is not None:
        for box in results[0].boxes:
            x1, y1, x2, y2 = map(int, box.xyxy[0].cpu().numpy())
            conf = float(box.conf[0])
            cls_id = int(box.cls[0])
            
            detections.append({
                "bbox": [x1, y1, x2, y2],
                "confidence": conf,
                "class_id": cls_id,
                "class_name": self.class_names[cls_id]
            })
    
    return detections
```

### 2.4 可视化

#### _draw_detections(image, detections)
```python
def _draw_detections(self, image, detections):
    """绘制检测框"""
    annotated = image.copy()
    
    for det in detections:
        x1, y1, x2, y2 = det["bbox"]
        conf = det["confidence"]
        cls_id = det["class_id"]
        color = self.class_colors[cls_id]
        
        # 绘制框
        cv2.rectangle(annotated, (x1, y1), (x2, y2), color, 2)
        
        # 绘制标签
        label = f"{det['class_name']} {conf:.2f}"
        (w, h), baseline = cv2.getTextSize(label, cv2.FONT_HERSHEY_SIMPLEX, 0.5, 1)
        cv2.rectangle(annotated, (x1, y1-h-5), (x1+w, y1), color, -1)
        cv2.putText(annotated, label, (x1, y1-5), 
                   cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0,0,0), 1)
    
    return annotated
```

**类别颜色**（BGR）：
```python
0: Noise              - (0, 0, 255)     红色
1: WiFi               - (255, 0, 255)   紫色
2: Bluetooth          - (0, 255, 255)   黄色
3: Video-transmission - (0, 255, 0)     绿色
4: Flight-control     - (0, 165, 255)   橙色
5: Fast-hopping       - (255, 255, 0)   青色
6: Control-pattern    - (255, 0, 0)     蓝色
```

## 3. ScanningController类

### 3.1 状态机

```
初始化
  ↓
enable_scanning? ──No──► DISABLED (固定频段显示)
  ↓ Yes
SCANNING (固定步长扫描)
  │
  ├─► 检测到飞控信号 ─────► LOCKED (动态跟踪)
  │                           │
  │                           ├─► 持续检测到 → 更新中心频率
  │                           └─► N帧无信号 ──┐
  │                                           ↓
  └─────────────────────────────────────── 返回SCANNING
```

### 3.2 初始化

```python
def __init__(self, state, data_processor):
    self.state = state
    self.data_processor = data_processor
    
    # 扫描参数
    self.enable_scanning = state.enable_scanning
    self.scan_bandwidth_mhz = state.scan_bandwidth_mhz  # 如200MHz
    self.overlap_ratio = state.overlap_ratio           # 如0.5
    self.control_lost_threshold = state.control_lost_threshold  # 如5帧
    
    # 计算窗口参数
    self._calculate_window_parameters()
    
    # 状态机
    self.scan_mode = ScanMode.SCANNING if self.enable_scanning else ScanMode.DISABLED
    self.current_window_index = 0
    self.locked_center_point = 0
    self.no_control_frame_count = 0
```

### 3.3 窗口计算

#### _calculate_window_parameters()
```python
def _calculate_window_parameters(self):
    """计算滑动窗口参数"""
    # 单窗口FFT点数
    # 例如200MHz → 200/100*512 = 1024点
    self.window_size = int(
        self.scan_bandwidth_mhz / 100 * self.state.fft_length
    )
    
    # 窗口步长（考虑重叠）
    # overlap=0.5 → step=512点
    self.window_step = int(self.window_size * (1 - self.overlap_ratio))
    
    # 总窗口数
    # (10240 - 1024) / 512 + 1 = 19个窗口
    total_points = self.data_processor.total_fft_length
    self.total_windows = (total_points - self.window_size) // self.window_step + 1
```

**示例计算**：
```
带宽=200MHz, FFT长度=512, 重叠率=0.5
→ window_size = 200/100*512 = 1024点
→ window_step = 1024 * 0.5 = 512点
→ total_windows = (10240-1024)/512+1 = 19窗口

窗口0: [0, 1024)      → 0-200MHz
窗口1: [512, 1536)    → 100-300MHz
窗口2: [1024, 2048)   → 200-400MHz
...
```

### 3.4 获取当前窗口

#### get_current_window_image()
```python
def get_current_window_image(self):
    """根据状态机返回当前窗口图像"""
    if not self.enable_scanning:
        # DISABLED模式：固定频段
        start_freq = self.state.start_frequency_mhz
        self.start_pt = int(start_freq * 10240 / 2000)
        self.end_pt = self.start_pt + self.window_size
    
    elif self.scan_mode == ScanMode.SCANNING:
        # SCANNING模式：固定步长
        self.start_pt, self.end_pt = self._get_window_range_scanning(
            self.current_window_index
        )
    
    elif self.scan_mode == ScanMode.LOCKED:
        # LOCKED模式：动态中心
        self.start_pt, self.end_pt = self._get_window_range_locked(
            self.locked_center_point
        )
    
    # 切片图像
    return self.data_processor.get_window_image(self.start_pt, self.end_pt)
```

#### _get_window_range_scanning(index)
```python
def _get_window_range_scanning(self, window_index):
    """SCANNING模式窗口范围"""
    start = window_index * self.window_step
    end = start + self.window_size
    end = min(end, 10240)
    return start, end
```

#### _get_window_range_locked(center)
```python
def _get_window_range_locked(self, center_point):
    """LOCKED模式窗口范围（居中）"""
    half = self.window_size // 2
    start = center_point - half
    end = center_point + half
    
    # 边界处理
    if start < 0:
        start, end = 0, self.window_size
    elif end > 10240:
        end, start = 10240, 10240 - self.window_size
    
    return start, end
```

### 3.5 状态机更新

#### update_state_machine(detections, image_width)
```python
def update_state_machine(self, detections, image_width):
    """根据检测结果更新状态"""
    if not self.enable_scanning or self.control_signal_class_id is None:
        return
    
    # 筛选飞控信号
    controls = [d for d in detections 
                if d["class_id"] == self.control_signal_class_id]
    
    has_signal = len(controls) > 0
    
    if self.scan_mode == ScanMode.SCANNING:
        if has_signal:
            # ✅ 进入LOCKED模式
            target = max(controls, key=lambda x: x["confidence"])
            bbox = target["bbox"]
            
            # 计算目标中心在图像中的位置
            center_x = (bbox[1] + bbox[3]) / 2
            
            # 转换为全局FFT点
            window_width = self.end_pt - self.start_pt
            center_pt = self.start_pt + int(
                center_x / image_width * window_width
            )
            
            self.scan_mode = ScanMode.LOCKED
            self.locked_center_point = center_pt
            self.no_control_frame_count = 0
            
            logger.info(f"🎯 锁定目标: {center_pt}点 "
                       f"({self.data_processor.get_point_to_frequency(center_pt):.1f}MHz)")
        else:
            # ➡️ 继续扫描
            self.current_window_index = (self.current_window_index + 1) % self.total_windows
    
    elif self.scan_mode == ScanMode.LOCKED:
        if has_signal:
            # ✅ 更新跟踪位置
            target = max(controls, key=lambda x: x["confidence"])
            bbox = target["bbox"]
            center_x = (bbox[1] + bbox[3]) / 2
            
            window_width = self.end_pt - self.start_pt
            new_center = self.start_pt + int(
                center_x / image_width * window_width
            )
            
            # 平滑更新（避免抖动）
            self.locked_center_point = int(
                self.locked_center_point * (1 - self.tracking_smoothing) +
                new_center * self.tracking_smoothing
            )
            
            self.no_control_frame_count = 0
        else:
            # ⚠️ 无信号，计数
            self.no_control_frame_count += 1
            
            if self.no_control_frame_count >= self.control_lost_threshold:
                # ❌ 丢失锁定 → 返回SCANNING
                self.current_window_index = (
                    self.locked_center_point // self.window_step
                )
                self.scan_mode = ScanMode.SCANNING
                
                logger.warning(f"⚠️ 锁定丢失，返回扫描模式（窗口{self.current_window_index}）")
```

### 3.6 状态查询

#### get_status()
```python
def get_status(self):
    """获取当前状态信息"""
    start_freq = self.data_processor.get_point_to_frequency(self.start_pt)
    end_freq = self.data_processor.get_point_to_frequency(self.end_pt)
    
    status = {
        "enabled": self.enable_scanning,
        "mode": self.scan_mode.value,
        "frequency_range_mhz": (start_freq, end_freq),
        "frequency_range_str": f"{start_freq:.1f}-{end_freq:.1f} MHz"
    }
    
    if self.scan_mode == ScanMode.SCANNING:
        status.update({
            "window_index": self.current_window_index + 1,
            "total_windows": self.total_windows,
            "scan_mode": "Scanning"
        })
    
    elif self.scan_mode == ScanMode.LOCKED:
        center_freq = self.data_processor.get_point_to_frequency(
            self.locked_center_point
        )
        status.update({
            "center_frequency_mhz": center_freq,
            "center_frequency_str": f"{center_freq:.2f} MHz",
            "no_signal_count": self.no_control_frame_count,
            "scan_mode": "Locked"
        })
    
    return status
```

## 6. 性能优化建议

1. **OpenVINO GPU**: 优先使用，速度提升3-5倍
2. **图像尺寸**: OpenVINO用512×512，PyTorch用640×640
3. **置信度阈值**: 过低会增加误检，过高会漏检
4. **扫描带宽**: 过大降低FPS，过小增加扫描时间
5. **重叠率**: 0.5为最佳平衡点
